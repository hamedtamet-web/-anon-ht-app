const swalBase = {

    customClass: {
        popup: 'swal-popup', title: 'swal-title',

    }

const toast = (icon, text, timer = 2000) => {
    if (typeof Swal === 'undefined') return;
    Swal.fire({
        ...swalBase, icon, text, timer,
        toast: true, position: 'bottom', showConfirmButton: false, timerProgressBar: true,
        iconColor: icon==='success' ? '#00e676' : icon==='error' ? '#f80566' : icon==='warning' ? '#ffcc00' : '#fff',
    });
};

const confirmSwal = (title, text, confirmTxt, dangerMode = false) => {
    if (typeof Swal === 'undefined') return Promise.resolve({ isConfirmed: confirm(text) });
    return Swal.fire({
        ...swalBase,
        icon: dangerMode ? 'warning' : 'question',
        iconColor: dangerMode ? '#ff4b4b' : '#f80566',
        title, text, showCancelButton: true,
        confirmButtonText: confirmTxt, cancelButtonText: t('cancelBtn'),
        customClass: {
            ...swalBase.customClass,
            confirmButton: dangerMode ? 'swal-confirm-danger' : 'swal-confirm'
        }
    });
};

/* ══ Helpers ══ */
const _esc = document.createElement('div');
function safe(s) { _esc.textContent = String(s ?? ''); return _esc.innerHTML; }

function timeAgo(ts) {
    if (!ts) return '';
    const m = Math.floor((Date.now() - ts.seconds * 1000) / 60000);
    if (m < 1)  return currentLang === 'ar' ? 'الآن' : 'Now';
    if (m < 60) return currentLang === 'ar' ? m+'د'  : m+'m';
    const h = Math.floor(m / 60);
    if (h < 24) return currentLang === 'ar' ? h+'س'  : h+'h';
    return currentLang === 'ar' ? Math.floor(h/24)+'ي' : Math.floor(h/24)+'d';
}

const forbiddenWords = ['كلمة_سيئة_1','كلمة_سيئة_2','badword1','badword2'];
function filterProfanity(text) {
    if (!text) return '';
    let r = text;
    forbiddenWords.forEach(w => { r = r.replace(new RegExp(w, 'gi'), '****'); });
    return r;
}

function safeImg(url, fallback = 'images/user.png') {
    if (!url) return fallback;
    try { return new URL(url).protocol === 'https:' ? url : fallback; } catch { return fallback; }
}

function applyLanguage(lang) {
    currentLang = lang;
    document.documentElement.lang = lang;
    document.documentElement.dir  = lang === 'ar' ? 'rtl' : 'ltr';

    /* Tabs */
    const tabAll      = document.getElementById('txt-tabAll');
    const tabPersonal = document.getElementById('txt-tabPersonal');
    const tabShout    = document.getElementById('txt-tabShout');
    if (tabAll)      tabAll.textContent      = t('tabAll');
    if (tabPersonal) tabPersonal.textContent = t('tabPersonal');
    if (tabShout)    tabShout.textContent    = t('tabShout');

    /* Bottom Nav */
    const navHome    = document.getElementById('nav-home');
    const navQues    = document.getElementById('nav-ques');
    const navQuiet   = document.getElementById('nav-quiet');
    const navGhost   = document.getElementById('nav-ghost');
    const navProfile = document.getElementById('nav-profile');
    if (navHome)    navHome.textContent    = t('nav_home');
    if (navQues)    navQues.textContent    = t('nav_qs');
    if (navQuiet)   navQuiet.textContent   = t('nav_skina');
    if (navGhost)   navGhost.textContent   = t('nav_tayf');
    if (navProfile) navProfile.textContent = t('nav_profile');

    /* زر اللغة نفسه */
    document.querySelectorAll('.lang-select-global').forEach(sel => { sel.value = lang; });
}

/* ══ استماع لتغيير اللغة من LangManager ══ */
document.addEventListener('langChanged', (e) => {
    const lang = e.detail?.lang || LangManager.get();
    applyLanguage(lang);
    renderAll();
});

/* ══════════════════════════════════════
   AUTH
══════════════════════════════════════ */
auth.onAuthStateChanged(user => {
    if (!user) { location.href = 'signin.html'; return; }
    currentUserId = user.uid;
    document.getElementById('navProfileLink').href = `profile.html?id=${user.uid}`;

    db.collection('users').doc(user.uid).onSnapshot(doc => {
        if (doc.exists) {
            document.getElementById('userCoins').textContent = (doc.data().anonCoins || 0).toFixed(2);
        }
    });

    /* ══ الأسئلة الشخصية ══ */
    db.collection('questions')
    .where('receiverId', '==', user.uid)
    .where('status', '==', 'pending')
    .orderBy('createdAt', 'desc')
    .onSnapshot(snap => {
        lastQuestionsSnap = snap;
        updateBadges();
        checkNewMessages();
        renderAll();
    }, err => console.error('Questions error:', err));

    /* ══ الشاوتات الواردة (من الآخرين فقط) ══ */
    db.collection('shouts')
        .orderBy('createdAt', 'desc')
        .limit(200)
        .onSnapshot(snap => {
            const filtered = {
                docs: snap.docs.filter(d => {
                    const data = d.data();
                    return data.ownerId !== user.uid && data.senderId !== user.uid;
                })
            };
            lastShoutsSnap = filtered;
            updateBadges();
            renderAll();
        }, err => console.error('Shouts error:', err));

    /* تطبيق اللغة المحفوظة عند أول تحميل */
    applyLanguage(LangManager.get());
});

/* ══════════════════════════════════════
   BADGES
══════════════════════════════════════ */
function updateBadges() {
    let qCount = 0, sCount = 0;
    if (lastQuestionsSnap) {
        lastQuestionsSnap.docs.forEach(d => { if (d.data().status === 'pending') qCount++; });
    }
    if (lastShoutsSnap) {
        lastShoutsSnap.docs.forEach(d => { if (!d.data().answeredAt) sCount++; });
    }
    const total = qCount + sCount;
    const navBadge = document.getElementById('qBadge');
    navBadge.textContent   = total;
    navBadge.style.display = total > 0 ? 'flex' : 'none';
    setBadge('badge-all',      total);
    setBadge('badge-personal', qCount);
    setBadge('badge-shout',    sCount);
}

function setBadge(id, count) {
    const el = document.getElementById(id);
    if (!el) return;
    el.textContent = count;
    el.classList.toggle('show', count > 0);
}

function checkNewMessages() {
    if (!lastQuestionsSnap) return;
    const pendingCount = lastQuestionsSnap.docs.filter(d => d.data().status === 'pending').length;
    if (!isInitialLoad && pendingCount > lastTotalCount) {
        document.getElementById('sndQuestion')?.play().catch(() => {});
        toast('success', t('newMsg'));
    }
    lastTotalCount = pendingCount;
    isInitialLoad  = false;
}

/* ══════════════════════════════════════
   RENDER
══════════════════════════════════════ */
function getMergedItems() {
    const items = [];
    if (lastQuestionsSnap) {
        lastQuestionsSnap.docs.forEach(doc => {
            const data = doc.data();
            items.push({ type:'question', id:doc.id, data, displayText: data.text||data.question||'', ts: data.createdAt?.seconds||0 });
        });
    }
    if (lastShoutsSnap) {
        lastShoutsSnap.docs.forEach(doc => {
            const data = doc.data();
            items.push({ type:'shout', id:doc.id, data, displayText: data.text||'', ts: data.createdAt?.seconds||0 });
        });
    }
    items.sort((a, b) => b.ts - a.ts);
    return items;
}

function renderAll() {
    const list  = document.getElementById('questionsList');
    let   items = getMergedItems();
    if (currentFilter === 'personal') items = items.filter(i => i.type === 'question');
    if (currentFilter === 'shout')    items = items.filter(i => i.type === 'shout');

    if (items.length === 0) {
        const emptyKey = currentFilter==='personal' ? 'emptyPersonal' : currentFilter==='shout' ? 'emptyShout' : 'emptyAll';
        const icons    = { personal:'fa-envelope', shout:'fa-broadcast-tower', all:'fa-inbox' };
        const iconKey  = currentFilter==='personal' ? 'personal' : currentFilter==='shout' ? 'shout' : 'all';
        list.innerHTML = `<div class="empty-state"><i class="fa-solid ${icons[iconKey]}" aria-hidden="true"></i><p>${safe(t(emptyKey))}</p></div>`;
        return;
    }

    const frag = document.createDocumentFragment();
    items.forEach(item => {
        const el = item.type === 'question' ? buildQuestionCard(item) : buildShoutCard(item);
        if (el) frag.appendChild(el);
    });
    list.innerHTML = '';
    list.appendChild(frag);
}

/* ══════════════════════════════════════
   بناء صف الاسم + الشارة
══════════════════════════════════════ */
function buildSenderElements(opts) {
    const { avatarUrl, name, username, isAdmin, isVerified, imgClass, wrapClass } = opts;

    const avatarWrap = document.createElement('div');
    avatarWrap.className = wrapClass + (isAdmin ? ' admin-glow' : isVerified ? ' verify-glow' : '');

    const img = document.createElement('img');
    img.src       = safeImg(avatarUrl);
    img.className = imgClass;
    img.alt       = '';
    img.loading   = 'lazy';
    img.onerror   = () => { img.src = 'images/user.png'; };
    avatarWrap.appendChild(img);

    const nameRow = document.createElement('div');
    nameRow.style.cssText = 'display:flex;align-items:center;gap:4px;flex-wrap:wrap;';

    const nameSpan = document.createElement('span');
    nameSpan.style.cssText = 'font-size:12px;font-weight:800;color:#fff;';
    nameSpan.textContent   = name || '';
    nameRow.appendChild(nameSpan);

    if (isAdmin) {
        const badge = document.createElement('img');
        badge.src       = 'images/admin-verified-logo.png';
        badge.className = 'badge-icon-sm admin-badge-sm';
        badge.alt       = '';
        nameRow.appendChild(badge);
    } else if (isVerified) {
        const badge = document.createElement('img');
        badge.src       = 'images/verified-logo.png';
        badge.className = 'badge-icon-sm verify-badge-sm';
        badge.alt       = '';
        nameRow.appendChild(badge);
    }

    const userLine = document.createElement('div');
    userLine.style.cssText = 'font-size:9px;color:var(--dim);margin-top:1px;';
    userLine.textContent   = username ? '@' + username : '';

    return { avatarWrap, nameRow, userLine };
}

/* ══════════════════════════════════════
   QUESTION CARD
══════════════════════════════════════ */
function buildQuestionCard(item) {
    const q       = item.data;
    const isAnon  = q.isAnonymous !== false;
    const isAdmin = (q.senderUsername || '').toUpperCase() === 'HT';
    const isVerif = !!(q.senderIsVerified || q.senderIsPremium);

    const div = document.createElement('div');
    div.className = 'ask-card';

    const body = document.createElement('div');
    body.className = 'ask-body';

    const label = document.createElement('span');
    label.className   = 'ask-label label-personal';
    label.textContent = t('personalLabel');

    const askText = document.createElement('p');
    askText.className   = 'ask-text';
    askText.textContent = filterProfanity(item.displayText);

    body.appendChild(label);
    body.appendChild(askText);

    if (!isAnon) {
        const senderRow = document.createElement('div');
        senderRow.className = 'sender-info';
        senderRow.style.cursor = 'pointer';
        senderRow.onclick = () => {
        if (q.senderId) {
        window.location.href = `profile.html?id=${q.senderId}`;
    }
};
        const { avatarWrap, nameRow, userLine } = buildSenderElements({
            avatarUrl:  q.senderAvatar,
            name:       q.senderName,
            username:   q.senderUsername,
            isAdmin,
            isVerified: isVerif,
            imgClass:   'sender-img',
            wrapClass:  'sender-avatar-wrap',
        });

        const textCol = document.createElement('div');
        textCol.className = 'sender-text-col';
        textCol.appendChild(nameRow);
        textCol.appendChild(userLine);

        senderRow.appendChild(avatarWrap);
        senderRow.appendChild(textCol);
        body.appendChild(senderRow);
    } else {
        const anonBadge = document.createElement('div');
        anonBadge.className = 'anon-shout-badge';
        anonBadge.innerHTML = `<i class="fa-solid fa-mask" aria-hidden="true"></i> ${safe(t('anonLabel'))}`;
        body.appendChild(anonBadge);
    }

    const footer = document.createElement('div');
    footer.className = 'ask-footer';

    const replyBtn = document.createElement('button');
    replyBtn.className = 'ask-btn reply';
    replyBtn.innerHTML = `<i class="fa-solid fa-feather-pointed" aria-hidden="true"></i> ${safe(t('replyPublish'))}`;
    replyBtn.onclick   = () => toggleBox(`qbox-${item.id}`);

    const deleteBtn = document.createElement('button');
    deleteBtn.className     = 'ask-btn icon-only';
    deleteBtn.setAttribute('aria-label', 'حذف');
    deleteBtn.innerHTML     = '<i class="fa-solid fa-trash-can" aria-hidden="true"></i>';
    deleteBtn.onclick       = () => handleDelete(item.id, 'question');

    const reportBtn = document.createElement('button');
    reportBtn.className     = 'ask-btn icon-only report-btn';
    reportBtn.setAttribute('aria-label', 'إبلاغ');
    reportBtn.innerHTML     = '<i class="fa-solid fa-flag" aria-hidden="true"></i>';
    reportBtn.onclick = () => handleReport(item.id, 'question', q.senderId || null);
    footer.appendChild(replyBtn);
    footer.appendChild(deleteBtn);
    footer.appendChild(reportBtn);

    const replyBox = document.createElement('div');
    replyBox.className = 'reply-box';
    replyBox.id        = `qbox-${item.id}`;

    const ta = document.createElement('textarea');
    ta.className   = 'reply-textarea';
    ta.id          = `qin-${item.id}`;
    ta.placeholder = t('replyPlaceholder');
    ta.maxLength   = 500;

    const sendBtn = document.createElement('button');
    sendBtn.className   = 'reply-send-btn';
    sendBtn.textContent = t('publishAnswer');
    sendBtn.onclick     = () => sendQuestionReply(item.id);

    replyBox.appendChild(ta);
    replyBox.appendChild(sendBtn);

    div.appendChild(body);
    div.appendChild(footer);
    div.appendChild(replyBox);
    return div;
}

/* ══════════════════════════════════════
   TOGGLE BOX
══════════════════════════════════════ */
function toggleBox(id) {
    const el = document.getElementById(id);
    if (!el) return;
    el.style.display = el.style.display === 'block' ? 'none' : 'block';
    if (el.style.display === 'block') {
        const ta = el.querySelector('textarea');
        if (ta) setTimeout(() => ta.focus(), 100);
    }
}

/* ══════════════════════════════════════
   SEND QUESTION REPLY
══════════════════════════════════════ */
async function sendQuestionReply(id) {
    const input   = document.getElementById(`qin-${id}`);
    const rawText = input?.value?.trim();
    if (!rawText) return toast('warning', t('replyEmpty'));
    const text = filterProfanity(rawText);

    try {
        const [qDoc, uDoc] = await Promise.all([
            db.collection('questions').doc(id).get(),
            db.collection('users').doc(currentUserId).get()
        ]);
        const qData = qDoc.data() || {};
        const uData = uDoc.data() || {};

        await db.collection('answers').add({
            questionId:   id,
            questionText: qData.text || qData.question || '',
            reply:        text,
            userId:       currentUserId,
            username:     uData.username || '',
            isVerified: !!(uData.username?.toUpperCase() === 'HT' || uData.isVerified || uData.isPremium),
            avatar:       uData.avatarUrl || '',
            answeredAt:   firebase.firestore.FieldValue.serverTimestamp(),
            likes:        []
        });
        await db.collection('questions').doc(id).update({ status: 'answered' });
        await db.collection('users').doc(currentUserId).update({
            answersCount: firebase.firestore.FieldValue.increment(1)
        });
        toast('success', t('replySuccess'));
        toggleBox(`qbox-${id}`);
        if (input) input.value = '';
    } catch (e) {
        console.error('sendQuestionReply:', e);
        toast('error', t('connError'));
    }
}

/* ══════════════════════════════════════
   SEND SHOUT REPLY
══════════════════════════════════════ */
async function sendShoutReply(shoutId) {
    const input   = document.getElementById(`sin-${shoutId}`);
    const rawText = input?.value?.trim();
    if (!rawText) return toast('warning', t('replyEmpty'));
    const text = filterProfanity(rawText);

    try {
        const [shoutDoc, uDoc] = await Promise.all([
            db.collection('shouts').doc(shoutId).get(),
            db.collection('users').doc(currentUserId).get()
        ]);
        const sData = shoutDoc.data() || {};
        const uData = uDoc.data() || {};

        await db.collection('shouts').doc(shoutId).collection('replies').add({
            senderId:     currentUserId,
            senderName:   uData.name || uData.username || '',
            senderAvatar: uData.avatarUrl || '',
            text,
            isAnswer:   true,
            likesCount: 0,
            likedBy:    [],
            createdAt:  firebase.firestore.FieldValue.serverTimestamp()
        });

        await db.collection('answers').add({
            questionId:   shoutId,
            questionText: sData.text || '',
            reply:        text,
            userId:       currentUserId,
            username:     uData.username || '',
            name:         uData.name || '',
            isVerified:   !!(uData.username?.toUpperCase() === 'HT' || uData.isVerified || uData.isPremium),
            avatar:       uData.avatarUrl || '',
            answeredAt:   firebase.firestore.FieldValue.serverTimestamp(),
            likes:        [],
            type:         'shout_reply'
        });

        await db.collection('shouts').doc(shoutId).update({
            repliesCount: firebase.firestore.FieldValue.increment(1)
        });

        await db.collection('users').doc(currentUserId).update({
            answersCount: firebase.firestore.FieldValue.increment(1)
        });

        toast('success', t('replySuccess'));
        toggleBox(`sbox-${shoutId}`);
        if (input) input.value = '';
        const card = document.getElementById(`sbox-${shoutId}`)?.closest('.shout-inbox-card');
        if (card) card.remove();
        lastShoutsSnap = { docs: lastShoutsSnap.docs.filter(d => d.id !== shoutId) };
        updateBadges();

    } catch (e) {
        console.error('sendShoutReply:', e);
        toast('error', t('connError'));
    }
}
/* ══════════════════════════════════════
   DELETE
══════════════════════════════════════ */
async function handleDelete(id, type) {
    const res = await confirmSwal(t('deleteTitle'), t('deleteDesc'), t('deleteBtn'), true);
    if (!res.isConfirmed) return;
    try {
        if (type === 'shout') {
            await db.collection('shouts').doc(id).delete();
        } else {
            await db.collection('questions').doc(id).delete();
        }
        toast('success', t('deleted'));
    } catch (e) {
        console.error('delete error:', e);
        toast('error', t('connError'));
    }
}

/* ══════════════════════════════════════
   REPORT
══════════════════════════════════════ */
async function handleReport(contentId, type, senderId) {
    if (typeof Swal === 'undefined') return;
    const { value: reason } = await Swal.fire({
        ...swalBase,
        title:            t('reportTitle'),
        input:            'textarea',
        inputPlaceholder: t('reportPlaceholder'),
        showCancelButton: true,
        confirmButtonText: t('reportSend'),
        cancelButtonText:  t('cancelBtn'),
    });
    if (!reason || !reason.trim()) return;

    try {
        await db.collection('reports').add({
            reportedBy:  currentUserId,
            contentId,
            contentType: type,
            reportedUser: senderId || null,
            reason:      reason.trim().slice(0, 500),
            createdAt:   firebase.firestore.FieldValue.serverTimestamp()
        });
        toast('success', t('reportSuccess'));
    } catch (e) {
        console.error('report error:', e);
        toast('error', t('connError'));
    }
}

   FILTER TABS

function setFilter(f, btn) {
    currentFilter = f;
    document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
    if (btn) btn.classList.add('active');
    renderAll();
}
